# Fırın Adisyon — Parça 4: Ödeme ve Hesap Bölme — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Masadan nakit ya da kartla ödeme almak. Dört yol var: tamamı, girilen tutar, kişi sayısına bölme, ürün seçerek. Kalan sıfırlanınca hesap kapanır ve kapanış özeti gösterilir.

**Architecture:** Sunucuda tek uç var: `POST /api/tables/{id}/payments`. Bu uç ya tutar (`AmountKurus`) ya da kalem listesi (`ItemIds`) alır. Kişi sayısına bölme ve para üstü istemcide hesaplanır, sunucuya yalnızca tutar gider. Ödemeler süreç içi bir kilit ve SQLite transaction'ı altında işlenir. İki cihaz aynı anda "tamamını öde" derse ikincisi reddedilir.

**Tech Stack:** Önceki parçalarla aynı.

**Ön koşul:** Parça 1–3 tamamlanmış olmalı.

## Global Constraints

- Hedef `net10.0`, `<Nullable>enable</Nullable>`; Node 24, TypeScript `strict`.
- Para her yerde **kuruş** cinsinden tam sayıdır.
- Ödeme tutarı 1 kuruş ile kalan tutar arasında olur. Fazla ödeme reddedilir; para üstü yalnızca ekranda hesaplanır, kaydedilmez.
- Kişi sayısına bölmede kuruş artığı son kişiye eklenir (100,00 ₺ / 3 = 33,33 + 33,33 + 33,34).
- Ürün seçerek ödemede kalemin `PaymentId` alanı dolar. O kalem artık değiştirilemez ve iptal edilemez (Parça 3 kuralı).
- Zaman her zaman DI'daki `TimeProvider` üzerinden alınır.
- Hatalar `DomainException(code, message, status)`. Arayüz metinleri Türkçe ve doğru diakritikli. Tarayıcı `alert/confirm` kullanılmaz. 394 px'te yatay kaydırma olmaz.

```csharp
using Firin.Api.Data;
using Firin.Api.Domain;
using Firin.Api.Realtime;

namespace Firin.Api.Orders;

public sealed class PaymentService(AppDb db, OrderService orders, IRealtime realtime)
{
    // Tek kasa sunucusu var: süreç içi kilit, aynı anda gelen ödemelerin kalan tutarı birlikte aşmasını önler.
    private static readonly SemaphoreSlim PaymentLock = new(1, 1);

    public async Task<PayResultDto> PayAsync(int tableId, PayRequest req, int userId)
    {
        var byAmount = req.AmountKurus is not null;
        var byItems = req.ItemIds is { Count: > 0 };
        if (byAmount == byItems || (req.ItemIds is not null && !byItems))
            throw new DomainException("odeme_gecersiz", "Ya tutar girin ya da ödenecek ürünleri seçin.");
        if (!Enum.IsDefined(req.Method))
            throw new DomainException("odeme_turu_gecersiz", "Ödeme türü geçersiz.");

        await PaymentLock.WaitAsync();
        try
        {
            await using var tx = await db.Database.BeginTransactionAsync();
            var order = await orders.LoadOpenOrderAsync(tableId)
                ?? throw new DomainException("hesap_yok", "Bu masada açık hesap yok.", StatusCodes.Status404NotFound);

            List<OrderItem> items = [];
            long amount;
            if (byItems)
            {
                var ids = req.ItemIds!.Distinct().ToList();
                items = order.Items.Where(i => ids.Contains(i.Id) && i.VoidedAt == null).ToList();
                if (items.Count != ids.Count)
                    throw new DomainException("kalem_yok", "Seçilen kalemlerden biri bu hesapta yok.", StatusCodes.Status404NotFound);
                if (items.Any(i => i.PaymentId is not null))
                    throw new DomainException("kalem_odendi", "Seçilen kalemlerden biri zaten ödenmiş.", StatusCodes.Status409Conflict);
                amount = items.Sum(i => i.LineTotalKurus);
            }
            else
            {
                amount = req.AmountKurus!.Value;
                if (amount <= 0) throw new DomainException("tutar_gecersiz", "Tutar 0'dan büyük olmalı.");
            }

            if (amount > OrderMath.Remaining(order))
                throw new DomainException("tutar_fazla", "Tutar kalan hesaptan fazla.", StatusCodes.Status409Conflict);

            var payment = new Payment { OrderId = order.Id, AmountKurus = amount, Method = req.Method, PaidAt = orders.Now, UserId = userId };
            order.Payments.Add(payment);
            await db.SaveChangesAsync();
            foreach (var i in items) i.PaymentId = payment.Id;
            orders.CloseIfSettled(order, userId);
            await db.SaveChangesAsync();
            await tx.CommitAsync();

            await realtime.TableChanged(tableId);
            var userName = (await db.Users.FindAsync(userId))?.Name ?? "?";
            return new PayResultDto(order.Id, order.Status == OrderStatus.Closed,
                new PaymentDto(payment.Id, payment.AmountKurus, payment.Method, payment.PaidAt, userName),
                await orders.GetTableAsync(tableId));
        }
        finally
        {
            PaymentLock.Release();
        }
    }
}
```

/** Tutarı kişi sayısına böler; kuruş artığı son kişiye eklenir. */
export function splitEqually(amount: number, people: number): number[] {
  if (people < 1 || amount <= 0) return [];
  const base = Math.floor(amount / people);
  const shares = Array<number>(people).fill(base);
  shares[people - 1] += amount - base * people;
  return shares;
}

export function changeDue(given: number, amount: number): number {
  return Math.max(0, given - amount);
}

/** 50, 100, 200, 500, 1000, 2000 ₺ */
const notes = [5000, 10000, 20000, 50000, 100000, 200000];

export function cashSuggestions(amount: number): number[] {
  const set = new Set<number>([amount]);
  for (const n of notes) {
    const v = Math.ceil(amount / n) * n;
    if (v > amount) set.add(v);
  }
  return [...set].sort((a, b) => a - b).slice(0, 4);
}

export function PaymentDialog({ table, disabled, onPaid, onClose }: Props) {

        <input inputMode="decimal" value={amountText} onChange={(e) => setAmountText(e.target.value)} autoFocus placeholder="50,00" />
        {k !== null && k > remaining && <span className="error">Tutar kalan hesaptan fazla.</span>}
      </label>
    );
  } else if (mode === 'split') {
    const left = Math.max(people - paidShares, 1);
    const shares = splitEqually(remaining, left);
    target = shares.length ? { amountKurus: shares[0] } : null;
    body = (
      <div className="split">
        <div className="stepper">
          <span>Kişi sayısı</span>
          <button aria-label="Azalt" disabled={people <= Math.max(2, paidShares + 1)} onClick={() => setPeople((p) => p - 1)}>−</button>
          <strong>{people}</strong>
          <button aria-label="Artır" disabled={people >= 50} onClick={() => setPeople((p) => p + 1)}>+</button>
        </div>
        <p className="muted">{paidShares} kişi ödedi, {left} kişi kaldı.</p>
        <ol className="shares">
          {shares.map((s, i) => <li key={i} className={i === 0 ? 'next' : ''}>{formatKurus(s)}</li>)}
        </ol>
      </div>
    );
  } else {
    const unpaid = order.items.filter((i) => !i.isPaid);
    const sum = unpaid.filter((i) => selected.includes(i.id)).reduce((s, i) => s + i.lineTotalKurus, 0);
    if (selected.length) target = { amountKurus: sum, itemIds: selected };
    body = (
      <>
        <ul className="pick-items">
          {unpaid.map((i) => (
            <li key={i.id}>
              <label className="check">
                <input type="checkbox" checked={selected.includes(i.id)}
                  onChange={(e) => setSelected((s) => (e.target.checked ? [...s, i.id] : s.filter((x) => x !== i.id)))} />
                <span className="pick-name">{i.productName} · {formatQuantity(i.unitType, i.quantity)}</span>
                <strong>{formatKurus(i.lineTotalKurus)}</strong>
              </label>
            </li>
          ))}
        </ul>
        <p className="pay-due">Seçilen: <strong>{formatKurus(sum)}</strong></p>
      </>
    );
  }

  return (
    <Modal title={`${table.name} · Ödeme`} onClose={onClose}>
      {cashFor ? (
        <CashStep target={cashFor} busy={busy} onBack={() => setCashFor(null)} onConfirm={(change) => send('Cash', cashFor, change)} />
      ) : (
        <>
          <div className="chips">
            {modes.map(([m, label]) => (
              <button key={m} className={m === mode ? 'chip active' : 'chip'} onClick={() => { setMode(m); setError(null); }}>{label}</button>
            ))}
          </div>
          {body}
          <div className="pay-buttons">
            <button className="primary" disabled={!target || busy} onClick={() => setCashFor(target)}>
              Nakit{target ? ` · ${formatKurus(target.amountKurus)}` : ''}
            </button>
            <button className="primary" disabled={!target || busy} onClick={() => target && send('Card', target)}>
              Kart{target ? ` · ${formatKurus(target.amountKurus)}` : ''}
            </button>
          </div>
        </>
      )}
      {error && <p className="error" role="alert">{error}</p>}
    </Modal>
  );
}

function CashStep({ target, busy, onBack, onConfirm }: {
  target: Target;
  busy: boolean;
  onBack: () => void;
  onConfirm: (change: number) => void;
}) {
  const [givenText, setGivenText] = useState('');
  const given = parseTlToKurus(givenText);
  const change = given !== null ? changeDue(given, target.amountKurus) : 0;
  const short = given !== null && given < target.amountKurus;

  return (
    <div className="cash-step">
      <p className="pay-due">Alınacak nakit: <strong>{formatKurus(target.amountKurus)}</strong></p>
      <div className="quick">
        {cashSuggestions(target.amountKurus).map((v) => (
          <button key={v} type="button" onClick={() => setGivenText((v / 100).toFixed(2).replace('.', ','))}>{formatKurus(v)}</button>
        ))}
      </div>
      <label className="form">
        Müşterinin verdiği (isteğe bağlı)
        <input inputMode="decimal" value={givenText} onChange={(e) => setGivenText(e.target.value)} placeholder="200,00" />
      </label>
      {short && <p className="error">Verilen para tutardan az.</p>}
      {change > 0 && <p className="change">Para üstü: <strong>{formatKurus(change)}</strong></p>}
      <div className="pay-buttons">
        <button type="button" onClick={onBack} disabled={busy}>Geri</button>
        <button type="button" className="primary" disabled={busy || short} onClick={() => onConfirm(change)}>Nakit ödemeyi kaydet</button>
      </div>
    </div>
  );
}

function ClosedSummary({ orderId, change, onClose }: { orderId: number; change: number; onClose: () => void }) {
  const navigate = useNavigate();
  const order = useQuery({ queryKey: ['order', orderId], queryFn: () => ordersApi.order(orderId) });
  return (
    <Modal title="Hesap kapandı" onClose={onClose}>
      <div className="closed-summary">
        <p className="ok-big">✓ Hesap kapandı</p>
        {order.data && (
          <>
            <p>Toplam: <strong>{formatKurus(order.data.totalKurus)}</strong></p>
            <ul className="payments">
              {order.data.payments.map((p) => (
                <li key={p.id}>{p.method === 'Cash' ? 'Nakit' : 'Kart'} · {formatKurus(p.amountKurus)}</li>
              ))}
            </ul>
          </>
        )}
        {change > 0 && <p className="change">Para üstü: <strong>{formatKurus(change)}</strong></p>}
        <button className="primary wide" onClick={() => navigate('/')}>Masalara dön</button>
        <button className="wide" onClick={onClose}>Bu masada kal</button>
      </div>
    </Modal>
  );
}
